65 explicit LinearProbe(int64_t activation_dim, DeviceBackend* backend, unsigned seed = 42)
66 : activation_dim_(activation_dim),
67 // Clamped only so a rejected dimension can't reach LinearModule/Shape and throw
68 // *their* message before the check below throws this class's own, clearer one --
69 // member initialization necessarily runs before the constructor body.
70 classifier_(activation_dim > 0 ? activation_dim : 1, 1, backend),
71 loss_(backend) {
72 if (activation_dim <= 0) {
73 throw std::invalid_argument("LinearProbe: activation_dim must be positive");
74 }
75 std::mt19937 rng(seed);
76 std::vector<float> weights(static_cast<size_t>(activation_dim));
77 for (float& w : weights) {
78 w = uniform_symmetric(rng, 0.01f);
79 }
80 classifier_.set_weight(weights);
81 classifier_.set_bias(std::vector<float>{uniform_symmetric(rng, 0.01f)});
82 }

102 float train_step(const Tensor& activation_batch, const Tensor& label_batch, OptimizerT& optimizer) {
103 validate_batch(activation_batch, label_batch, "LinearProbe::train_step");
104
105 optimizer.zero_grad(classifier_);
106
107 Tensor logits = classifier_.forward(activation_batch);
108 const float loss_value = loss_.forward(logits, label_batch);
109
110 Tensor grad_logits = loss_.backward();
111 (void)classifier_.backward(grad_logits); // the activations have no upstream to receive this
112
113 optimizer.step(classifier_);
114 return loss_value;

137 [[nodiscard]] float accuracy(const Tensor& activation_batch, const Tensor& label_batch) const {
138 validate_batch(activation_batch, label_batch, "LinearProbe::accuracy");
139
140 const Tensor logits = classifier_.forward(activation_batch);
141 const int64_t n = logits.numel();
142 int64_t correct = 0;
143 for (int64_t i = 0; i < n; ++i) {
144 const bool predicted_positive = logits.data()[i] >= 0.0f;
145 const bool labeled_positive = label_batch.data()[i] >= 0.5f;
146 if (predicted_positive == labeled_positive) {

148 }
149 }
150 return static_cast<float>(correct) / static_cast<float>(n);

168 static float uniform_symmetric(std::mt19937& rng, float scale) {
169 // std::uniform_real_distribution's mapping is implementation-defined, so identical
170 // seeds would not give identical weights across standard libraries. This mapping is
171 // fixed here, making a seeded probe reproducible everywhere.
172 const float unit = static_cast<float>(rng() - std::mt19937::min()) /
173 static_cast<float>(std::mt19937::max() - std::mt19937::min() + 1ull);
174 return (unit * 2.0f - 1.0f) * scale;
175 }

192 void validate_batch(const Tensor& activation_batch, const Tensor& label_batch, const char* method) const {
193 const std::string where(method);
194 if (activation_batch.rank() != 2) {
195 throw std::invalid_argument(where + ": activation_batch must be rank-2 (N, activation_dim)");
196 }
197 if (label_batch.rank() != 2 || label_batch.shape().dim(1) != 1) {
198 throw std::invalid_argument(where + ": label_batch must be rank-2 (N, 1)");
199 }
200 if (activation_batch.shape().dim(1) != activation_dim_) {
201 throw std::invalid_argument(where + ": activation_batch width must equal activation_dim");
202 }
203 if (activation_batch.shape().dim(0) != label_batch.shape().dim(0)) {
204 throw std::invalid_argument(where + ": activation_batch and label_batch must have the same batch size");
205 }
206 if (activation_batch.shape().dim(0) <= 0) {
207 throw std::invalid_argument(where + ": batch must not be empty");
208 }
209 }
