95template <class Json>
96void to_json(Json& j, const JsonValue& value) {
97 switch (value.type()) {
98 case JsonValue::Type::Null:
99 j = nullptr;
100 return;
101 case JsonValue::Type::Bool:
102 j = value.as_bool();
103 return;
104 case JsonValue::Type::Number: {
105 const std::string& text = value.number_text();
106 if (text.find_first_of(".eE") == std::string::npos) {
107 try {
108 j = static_cast<std::int64_t>(value.as_int64());
109 return;
110 } catch (const std::invalid_argument&) {
111 }
112 }
113 j = value.as_double();
114 return;
115 }
116 case JsonValue::Type::String:
117 j = value.as_string();
118 return;
119 case JsonValue::Type::Array: {
120 j = Json::array();
121 for (const JsonValue& element : value.as_array()) {
122 Json converted;
123 to_json(converted, element);
124 j.push_back(std::move(converted));
125 }
126 return;
127 }
128 case JsonValue::Type::Object: {
129 j = Json::object();
130 for (const auto& [key, member] : value.as_object()) {
131 Json converted;
132 to_json(converted, member);
133 j[key] = std::move(converted);
134 }
135 return;
136 }
137 }
138}
