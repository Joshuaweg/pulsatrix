484 virtual void adam_step_multi(const AdamTensorStep* tensors, size_t count, float beta1, float beta2, float eps) {
485 for (size_t t = 0; t < count; ++t) {
486 const AdamTensorStep& s = tensors[t];
487 if (s.n == 0) continue;
488 if (s.decay != 1.0f) axpby(s.decay, s.param, 0.0f, s.param, s.param, s.n);
489 const float* grad = s.grad;
490 void* decayed = nullptr;
491 if (s.l2 != 0.0f) {
492 decayed = allocate(s.n * sizeof(float));
493 axpby(s.l2, s.param, 1.0f, s.grad, static_cast<float*>(decayed), s.n);
494 grad = static_cast<const float*>(decayed);
495 }
496 adam_step(s.param, grad, s.m, s.v, s.n, s.lr, beta1, beta2, eps, s.bias_correction1, s.bias_correction2);
497 free(decayed);
498 }
