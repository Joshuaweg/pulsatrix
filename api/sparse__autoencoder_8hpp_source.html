97 SparseAutoencoder(int64_t dim, int64_t hidden_dim, float l1_lambda, DeviceBackend* backend, unsigned seed)
98 : dim_(dim),
99 hidden_dim_(hidden_dim),
100 l1_lambda_(l1_lambda),
101 backend_(backend),
102 // Clamped only so a rejected dimension can't reach LinearModule/Shape and throw
103 // *their* message before the check below throws this class's own, clearer one --
104 // member initialization necessarily runs before the constructor body.
105 encoder_(dim > 0 ? dim : 1, hidden_dim > 0 ? hidden_dim : 1, backend),
106 relu_(backend),
107 decoder_(hidden_dim > 0 ? hidden_dim : 1, dim > 0 ? dim : 1, backend),
108 loss_(backend) {
109 if (dim <= 0) {
110 throw std::invalid_argument("SparseAutoencoder: dim must be positive");
111 }
112 if (hidden_dim <= 0) {
113 throw std::invalid_argument("SparseAutoencoder: hidden_dim must be positive");
114 }
115 if (l1_lambda < 0.0f) {
116 throw std::invalid_argument("SparseAutoencoder: l1_lambda must be non-negative");
117 }
118
119 std::mt19937 rng(seed);
120 // Fan-in scaling: each layer's init range shrinks with its own input width, so the
121 // pre-activation magnitude at step 0 stays O(1) regardless of dim/hidden_dim rather
122 // than growing with the layer's width (which for a deliberately overcomplete SAE
123 // would otherwise saturate the decoder's input from the first step).
124 init_layer(encoder_, rng, 1.0f / std::sqrt(static_cast<float>(dim)));
125 init_layer(decoder_, rng, 1.0f / std::sqrt(static_cast<float>(hidden_dim)));
126 normalize_decoder();
127 }

199 FeaturizerLoss loss_and_backward(const Tensor& input_batch, std::vector<float>* codes = nullptr) override {
200 validate_batch(input_batch, "SparseAutoencoder::loss_and_backward");
201 const Tensor hidden = relu_.forward(encoder_.forward(input_batch));
202 const Tensor reconstruction = decoder_.forward(hidden);
203 FeaturizerLoss loss;
204 loss.reconstruction = loss_.forward(reconstruction, input_batch);
205 const Tensor grad_reconstruction = loss_.backward();
206 Tensor grad_hidden = decoder_.backward(grad_reconstruction);
207 const float batch_size = static_cast<float>(input_batch.shape().dim(0));
208 Tensor l1_grad(grad_hidden.shape(), backend_, grad_hidden.device());
209 l1_grad.fill(l1_lambda_ / batch_size);
210 grad_hidden.accumulate(l1_grad);
211 const Tensor grad_pre_activation = relu_.backward(grad_hidden);
212 (void)encoder_.backward(grad_pre_activation); // the input has no upstream to receive this
213 const std::vector<float> h = hidden.to_host_vector();
214 double sum = 0.0;
215 for (float v : h) sum += v;
216 loss.sparsity = static_cast<float>(l1_lambda_ * sum / batch_size);
217 loss.total = loss.reconstruction + loss.sparsity;
218 if (codes != nullptr) *codes = h;
219 return loss;
220 }
224 void normalize_decoder() override {
225 std::vector<float> dec = decoder_.weight().to_host_vector(), enc = encoder_.weight().to_host_vector(),
226 bias = encoder_.bias().to_host_vector();
227 for (int64_t i = 0; i < hidden_dim_; ++i) {
228 double sq = 0.0;
229 for (int64_t j = 0; j < dim_; ++j) sq += static_cast<double>(dec[static_cast<size_t>(i * dim_ + j)]) * dec[static_cast<size_t>(i * dim_ + j)];
230 const double norm = std::sqrt(sq);
231 if (norm == 0.0) continue;
232 for (int64_t j = 0; j < dim_; ++j) dec[static_cast<size_t>(i * dim_ + j)] = static_cast<float>(dec[static_cast<size_t>(i * dim_ + j)] / norm);
233 for (int64_t j = 0; j < dim_; ++j) enc[static_cast<size_t>(j * hidden_dim_ + i)] = static_cast<float>(enc[static_cast<size_t>(j * hidden_dim_ + i)] * norm);
234 bias[static_cast<size_t>(i)] = static_cast<float>(bias[static_cast<size_t>(i)] * norm);
235 }
236 decoder_.set_weight(dec);
237 encoder_.set_weight(enc);
238 encoder_.set_bias(bias);
239 }

404 static void init_layer(LinearModule& layer, std::mt19937& rng, float scale) {
405 std::vector<float> weights(static_cast<size_t>(layer.weight().numel()));
406 for (float& w : weights) {
407 w = uniform_symmetric(rng, scale);
408 }
409 layer.set_weight(weights);
410 layer.set_bias(std::vector<float>(static_cast<size_t>(layer.bias().numel()), 0.0f));
411 }

414 static float uniform_symmetric(std::mt19937& rng, float scale) {
415 // std::uniform_real_distribution's mapping is implementation-defined, so identical
416 // seeds would not give identical weights across standard libraries. This mapping is
417 // fixed here, making a seeded SAE reproducible everywhere -- the same convention
418 // LinearProbe already uses.
419 const float unit = static_cast<float>(rng() - std::mt19937::min()) /
420 static_cast<float>(std::mt19937::max() - std::mt19937::min() + 1ull);
421 return (unit * 2.0f - 1.0f) * scale;
422 }
