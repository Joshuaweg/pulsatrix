84 SparseAutoencoder(int64_t dim, int64_t hidden_dim, float l1_lambda, DeviceBackend* backend, unsigned seed = 42)
85 : dim_(dim),
86 hidden_dim_(hidden_dim),
87 l1_lambda_(l1_lambda),
88 backend_(backend),
89 // Clamped only so a rejected dimension can't reach LinearModule/Shape and throw
90 // *their* message before the check below throws this class's own, clearer one --
91 // member initialization necessarily runs before the constructor body.
92 encoder_(dim > 0 ? dim : 1, hidden_dim > 0 ? hidden_dim : 1, backend),
93 relu_(backend),
94 decoder_(hidden_dim > 0 ? hidden_dim : 1, dim > 0 ? dim : 1, backend),
95 loss_(backend) {
96 if (dim <= 0) {
97 throw std::invalid_argument("SparseAutoencoder: dim must be positive");
98 }
99 if (hidden_dim <= 0) {
100 throw std::invalid_argument("SparseAutoencoder: hidden_dim must be positive");
101 }
102 if (l1_lambda < 0.0f) {
103 throw std::invalid_argument("SparseAutoencoder: l1_lambda must be non-negative");
104 }
105
106 std::mt19937 rng(seed);
107 // Fan-in scaling: each layer's init range shrinks with its own input width, so the
108 // pre-activation magnitude at step 0 stays O(1) regardless of dim/hidden_dim rather
109 // than growing with the layer's width (which for a deliberately overcomplete SAE
110 // would otherwise saturate the decoder's input from the first step).
111 init_layer(encoder_, rng, 1.0f / std::sqrt(static_cast<float>(dim)));
112 init_layer(decoder_, rng, 1.0f / std::sqrt(static_cast<float>(hidden_dim)));
113 }

154 template <typename OptimizerT>
155 float train_step(const Tensor& input_batch, OptimizerT& optimizer) {
156 validate_batch(input_batch, "SparseAutoencoder::train_step");
157
158 optimizer.zero_grad(encoder_);
159 optimizer.zero_grad(decoder_);
160
161 const Tensor hidden = relu_.forward(encoder_.forward(input_batch));
162 const Tensor reconstruction = decoder_.forward(hidden);
163 const float reconstruction_loss = loss_.forward(reconstruction, input_batch);
164
165 const Tensor grad_reconstruction = loss_.backward();
166 Tensor grad_hidden = decoder_.backward(grad_reconstruction);
167
168 const float batch_size = static_cast<float>(input_batch.shape().dim(0));
169 Tensor l1_grad(grad_hidden.shape(), backend_, grad_hidden.device());
170 l1_grad.fill(l1_lambda_ / batch_size);
171 grad_hidden.accumulate(l1_grad);
172
173 const Tensor grad_pre_activation = relu_.backward(grad_hidden);
174 (void)encoder_.backward(grad_pre_activation); // the input has no upstream to receive this
175
176 optimizer.step(encoder_);
177 optimizer.step(decoder_);
178 // relu_ has no parameters (Module::parameters() default) -- stepping it would be a
179 // safe no-op but would imply there is something to update.
180 return reconstruction_loss;
181 }

228 [[nodiscard]] float reconstruction_error(const Tensor& input_batch) const {
229 validate_batch(input_batch, "SparseAutoencoder::reconstruction_error");
230
231 const Tensor reconstruction = reconstruct(input_batch);
232 const int64_t n = reconstruction.numel();
233 float sum_squared = 0.0f;
234 for (int64_t i = 0; i < n; ++i) {
235 const float diff = reconstruction.data()[i] - input_batch.data()[i];
236 sum_squared += diff * diff;
237 }
238 return sum_squared / static_cast<float>(n);
239 }

299 static void init_layer(LinearModule& layer, std::mt19937& rng, float scale) {
300 std::vector<float> weights(static_cast<size_t>(layer.weight().numel()));
301 for (float& w : weights) {
302 w = uniform_symmetric(rng, scale);
303 }
304 layer.set_weight(weights);
305 layer.set_bias(std::vector<float>(static_cast<size_t>(layer.bias().numel()), 0.0f));
306 }

309 static float uniform_symmetric(std::mt19937& rng, float scale) {
310 // std::uniform_real_distribution's mapping is implementation-defined, so identical
311 // seeds would not give identical weights across standard libraries. This mapping is
312 // fixed here, making a seeded SAE reproducible everywhere -- the same convention
313 // LinearProbe already uses.
314 const float unit = static_cast<float>(rng() - std::mt19937::min()) /
315 static_cast<float>(std::mt19937::max() - std::mt19937::min() + 1ull);
316 return (unit * 2.0f - 1.0f) * scale;
317 }
