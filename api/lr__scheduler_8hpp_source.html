52 [[nodiscard]] float multiplier(int64_t step) const {
53 if (step < 0) {
54 throw std::invalid_argument("LRSchedule::multiplier: step must be >= 0");
55 }
56 if (step < warmup_) {
57 return static_cast<float>(static_cast<double>(step) / static_cast<double>(warmup_));
58 }
59 const double span = static_cast<double>(std::max<int64_t>(1, total_ - warmup_));
60 switch (kind_) {
61 case Kind::Constant:
62 return 1.0f;
63 case Kind::Linear:
64 return static_cast<float>(std::max(0.0, static_cast<double>(total_ - step) / span));
65 case Kind::Cosine: {
66 constexpr double kPi = 3.14159265358979323846;
67 const double progress = std::min(1.0, static_cast<double>(step - warmup_) / span);
68 return static_cast<float>(min_ratio_ + (1.0 - min_ratio_) * 0.5 * (1.0 + std::cos(kPi * progress)));
69 }
70 }
71 return 1.0f;
72 }
73
74private:
75 enum class Kind { Constant, Linear, Cosine };
76
77 LRSchedule(Kind kind, int64_t warmup, int64_t total, float min_ratio)
78 : kind_(kind), warmup_(warmup), total_(total), min_ratio_(min_ratio) {}

101public:
102 LRScheduler(Optimizer& optimizer, LRSchedule schedule)
103 : optimizer_(optimizer), schedule_(schedule), base_(optimizer.learning_rate()) {
104 for (const auto& group : optimizer.param_groups()) {
105 group_bases_.push_back(group.learning_rate);
106 }
107 apply();
108 }
109
111 void step() {
112 ++step_;
113 apply();
114 }
115
117 [[nodiscard]] int64_t last_step() const { return step_; }
118
121 void set_last_step(int64_t step) {
122 if (step < 0) {
123 throw std::invalid_argument("LRScheduler::set_last_step: step must be >= 0");
124 }
125 step_ = step;
126 apply();
127 }
128
129private:
130 void apply() {
131 auto& groups = optimizer_.param_groups();
132 if (groups.size() != group_bases_.size()) {
133 throw std::logic_error("LRScheduler: the optimizer's parameter groups changed after the scheduler was built");
134 }
135 const float m = schedule_.multiplier(step_);
136 optimizer_.set_learning_rate(base_ * m);
137 for (size_t i = 0; i < groups.size(); ++i) {
138 groups[i].learning_rate = group_bases_[i] * m;
139 }
140 }
141
142 Optimizer& optimizer_;
143 LRSchedule schedule_;
144 float base_;
145 std::vector<float> group_bases_;
146 int64_t step_ = 0;
147};
