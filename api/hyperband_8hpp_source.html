47inline std::vector<HyperbandBracket> ComputeHyperbandBrackets(int max_resource, double eta) {
48 if (max_resource <= 0) {
49 throw std::invalid_argument("ComputeHyperbandBrackets: max_resource must be positive");
50 }
51 if (eta <= 1.0) {
52 throw std::invalid_argument("ComputeHyperbandBrackets: eta must be > 1.0");
53 }
54
55 int s_max = static_cast<int>(std::floor(std::log(static_cast<double>(max_resource)) / std::log(eta)));
56 double b = static_cast<double>(s_max + 1) * static_cast<double>(max_resource);
57
58 std::vector<HyperbandBracket> brackets;
59 for (int s = s_max; s >= 0; --s) {
60 double eta_pow_s = std::pow(eta, s);
61 size_t num_configs =
62 std::max<size_t>(1, static_cast<size_t>(std::ceil((b / max_resource) * (eta_pow_s / (s + 1)))));
63 int initial_budget = std::max(1, static_cast<int>(std::round(max_resource / eta_pow_s)));
64 brackets.push_back(HyperbandBracket{s, num_configs, initial_budget});
65 }
66 return brackets;
67}
