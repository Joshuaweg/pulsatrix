97 const std::vector<float> grad = grad_tensor.to_host_vector();
98 auto at4 = [&](const std::vector<float>& v, int64_t n, int64_t c, int64_t h, int64_t w) {
99 return v[static_cast<size_t>(((n * channels + c) * height + h) * width + w)];

105 for (int64_t c = 0; c < channels; ++c) {
106 float sum = 0.0f;
107 for (int64_t h = 0; h < height; ++h) {
108 for (int64_t w = 0; w < width; ++w) {
109 sum += at4(grad, n, c, h, w);
110 }
111 }
112 alpha[static_cast<size_t>(c)] = sum / static_cast<float>(height * width);
113 }
114
115 for (int64_t h = 0; h < height; ++h) {
116 for (int64_t w = 0; w < width; ++w) {
117 float value = 0.0f;
118 for (int64_t c = 0; c < channels; ++c) {
119 value += alpha[static_cast<size_t>(c)] * at4(activation, n, c, h, w);
120 }
121 cam[static_cast<size_t>((n * height + h) * width + w)] = value > 0.0f ? value : 0.0f;
122 }
123 }
