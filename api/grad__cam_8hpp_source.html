55 [[nodiscard]] Attribution explain(ExplainerContext& ctx, const Tensor& input, int64_t target_index,
56 DeviceBackend* backend) const {
57 Tensor output = ctx.forward_pass(input);
58 // External boundary (Mission 2, finding 15 systemic sweep; rank check added by
59 // campaign_exai_dl_library_batch_dimension_support) -- escalated from
60 // PULSATRIX_ASSERT-only.
61 if (output.rank() != 2) {
62 throw std::invalid_argument("GradCAM::explain: network output must be rank-2 (N, num_classes)");
63 }
64 if (target_index < 0 || target_index >= output.shape().dim(1)) {
65 throw std::invalid_argument("GradCAM::explain: target_index out of range");
66 }
67 int64_t N = output.shape().dim(0);
68
69 Tensor seed(output.shape(), backend);
70 seed.fill(0.0f);
71 for (int64_t n = 0; n < N; ++n) {
72 seed.at({n, target_index}) = 1.0f;
73 }
74 (void)ctx.backward_pass(seed);
75
76 std::vector<NodeId> conv_nodes = ctx.graph().nodes_by_op_type(OpType::Conv);
77 // External boundary -- whether this ExplainerContext was built with a Conv layer
78 // is a caller-configuration fact, not an internal invariant this library controls.
79 if (conv_nodes.empty()) {
80 throw std::invalid_argument("GradCAM::explain: graph has no Conv layer");
81 }
82 NodeId target_node = conv_nodes.back();
83
84 const Tensor& activation = ctx.activation(target_node);
85 const Tensor& grad = ctx.gradient(target_node);
86
87 int64_t channels = activation.shape().dim(1);
88 int64_t height = activation.shape().dim(2);
89 int64_t width = activation.shape().dim(3);
90
91 Tensor cam(Shape({N, height, width}), backend);
92 for (int64_t n = 0; n < N; ++n) {
93 std::vector<float> alpha(static_cast<size_t>(channels), 0.0f);
94 for (int64_t c = 0; c < channels; ++c) {
95 float sum = 0.0f;
96 for (int64_t h = 0; h < height; ++h) {
97 for (int64_t w = 0; w < width; ++w) {
98 sum += grad.at({n, c, h, w});
99 }
100 }
101 alpha[static_cast<size_t>(c)] = sum / static_cast<float>(height * width);
102 }
103
104 for (int64_t h = 0; h < height; ++h) {
105 for (int64_t w = 0; w < width; ++w) {
106 float value = 0.0f;
107 for (int64_t c = 0; c < channels; ++c) {
108 value += alpha[static_cast<size_t>(c)] * activation.at({n, c, h, w});
109 }
110 cam.at({n, h, w}) = value > 0.0f ? value : 0.0f;
111 }
112 }
113 }
114
115 return Attribution{"grad_cam", std::move(cam),
116 {{"target_index", std::to_string(target_index)},
117 {"layer_node_id", std::to_string(target_node)}}};
118 }
