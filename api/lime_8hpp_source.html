139 [[nodiscard]] static double local_fidelity(const std::vector<std::vector<float>>& samples,
140 const std::vector<float>& targets, const std::vector<float>& weights,
141 const std::vector<float>& coefficients) {
142 double weight_sum = 0.0, mean = 0.0;
143 for (size_t s = 0; s < samples.size(); ++s) {
144 weight_sum += weights[s];
145 mean += static_cast<double>(weights[s]) * targets[s];
146 }
147 if (weight_sum <= 0.0) return 0.0;
148 mean /= weight_sum;
149 double residual = 0.0, total = 0.0;
150 for (size_t s = 0; s < samples.size(); ++s) {
151 double fit = 0.0;
152 for (size_t i = 0; i < coefficients.size(); ++i) fit += static_cast<double>(coefficients[i]) * samples[s][i];
153 const double e = targets[s] - fit, d = targets[s] - mean;
154 residual += weights[s] * e * e;
155 total += weights[s] * d * d;
156 }
157 if (total == 0.0) return residual == 0.0 ? 1.0 : 0.0;
158 return 1.0 - residual / total;
159 }
