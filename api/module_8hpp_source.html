236 void set_requires_grad(bool requires_grad, const std::string& prefix = "") {
237 if (prefix.empty()) {
238 // parameters(), not named_parameters(): also reaches a legacy module that has no names.
239 for (ParamRef p : parameters()) {
240 p.value->set_requires_grad(requires_grad);
241 }
242 return;
243 }
244 std::vector<Tensor*> selected;
245 for (const NamedParamRef& p : named_parameters()) {
246 if (p.name == prefix || p.name.rfind(prefix + ".", 0) == 0) {
247 selected.push_back(p.ref.value);
248 }
249 }
250 if (selected.empty()) {

252 }
