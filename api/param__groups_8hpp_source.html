92 [[nodiscard]] std::vector<Assignment> resolve(Module& module, float default_learning_rate,
93 float default_weight_decay) const {
94 for (const ParamGroup& g : groups_) {
95 check(g);
96 }
97 std::unordered_map<const Tensor*, const ParamGroup*> chosen;
98 if (!groups_.empty()) {
99 for (const NamedParamRef& p : module.named_parameters()) {
100 for (const ParamGroup& g : groups_) {
101 if (g.selects(p.name, *p.ref.value)) {
102 chosen.emplace(p.ref.value, &g);
103 break;
104 }
105 }
106 }
107 }
108 std::vector<Assignment> out;
109 for (ParamRef p : module.parameters()) {
110 auto it = chosen.find(p.value);
111 if (it == chosen.end()) {
112 out.push_back({p, default_learning_rate, default_weight_decay});
113 } else {
114 out.push_back({p, it->second->learning_rate, it->second->weight_decay});
115 }
116 }
117 return out;
118 }
119
120private:
121 static void check(const ParamGroup& g) {
122 if (!g.selects) {
123 throw std::invalid_argument("parameter group \"" + g.name + "\" has no selector");
124 }
125 check_optimizer_setting(g.learning_rate, "parameter group \"" + g.name + "\" learning_rate");
126 check_optimizer_setting(g.weight_decay, "parameter group \"" + g.name + "\" weight_decay");
127 }

129 std::vector<ParamGroup> groups_;
