73 Tensor forward_pass(const Tensor& input) {
74 last_forward_was_patched_ = false;
75 graph_ = ComputationGraph{};
76 autograd_ = Autograd{};
77 activations_.clear();
78
79 input_node_ = graph_.add_node(OpType::Elementwise, input.shape(), "input");
80 activations_.emplace(input_node_, Tensor(input));
81
82 Tensor current = input;
83 NodeId current_node = input_node_;
84 for (Module* module : modules_) {
85 auto [output, node_id] = module->forward_traced(current, current_node, graph_, autograd_);
86 activations_.emplace(node_id, Tensor(output));
87 current = std::move(output);
88 current_node = node_id;
89 }
90 output_node_ = current_node;
91 return current;
92 }

131 Tensor forward_pass_with_patch(const Tensor& input, NodeId patch_node_id, const Tensor& patch_value) {
132 last_forward_was_patched_ = true;
133 if (patch_node_id > modules_.size()) {
134 throw std::invalid_argument("ExplainerContext::forward_pass_with_patch: patch_node_id out of range");
135 }
136
137 ComputationGraph graph;
138 Autograd autograd;
139 std::unordered_map<NodeId, Tensor> activations;
140
141 NodeId input_node = graph.add_node(OpType::Elementwise, input.shape(), "input");
142
143 Tensor current = input;
144 NodeId current_node = input_node;
145 if (patch_node_id == input_node) {
146 current = check_patch_shape(current, patch_value);
147 }
148 activations.emplace(input_node, Tensor(current));
149
150 for (Module* module : modules_) {
151 auto [output, node_id] = module->forward_traced(current, current_node, graph, autograd);
152 if (node_id == patch_node_id) {
153 output = check_patch_shape(output, patch_value);
154 }
155 activations.emplace(node_id, Tensor(output));
156 current = std::move(output);
157 current_node = node_id;
158 }
159
160 graph_ = std::move(graph);
161 autograd_ = std::move(autograd);
162 activations_ = std::move(activations);
163 input_node_ = input_node;
164 output_node_ = current_node;
165 return current;
166 }

185 Tensor backward_pass(const Tensor& output_grad) {
186 if (last_forward_was_patched_) {
187 throw std::logic_error(
188 "ExplainerContext::backward_pass: the most recent forward pass was "
189 "forward_pass_with_patch(); a patched activation is a constant substitution, not a "
190 "differentiable function of the input, so gradients through it would be silently wrong. "
191 "Run an unpatched forward_pass() before backward_pass().");
192 }
193 autograd_.backward(graph_, output_node_, output_grad);
194 return Tensor(autograd_.gradient(input_node_));
195 }

209 Tensor relevance_pass(const Tensor& output_relevance, const LRPRuleConfig& config) {
210 if (last_forward_was_patched_) {
211 throw std::logic_error(
212 "ExplainerContext::relevance_pass: the most recent forward pass was "
213 "forward_pass_with_patch(); relevance through a patched activation would describe a "
214 "computation the input did not produce. Run an unpatched forward_pass() first.");
215 }
216 Tensor relevance = output_relevance;
217 for (auto it = modules_.rbegin(); it != modules_.rend(); ++it) {
218 relevance = (*it)->propagate_relevance(relevance, config);
219 }

265 [[nodiscard]] ActivationSnapshot activation_snapshot() const {
266 std::vector<NodeId> node_ids = graph_.topological_order();
267
268 std::unordered_map<NodeId, Tensor> activations;
269 std::unordered_map<NodeId, ActivationSnapshot::NodeMetadata> metadata;
270 for (NodeId id : node_ids) {
271 auto it = activations_.find(id);
272 PULSATRIX_ASSERT(it != activations_.end());
273 activations.emplace(id, Tensor(it->second));
274 const Node& node = graph_.node(id);
275 metadata.emplace(id, ActivationSnapshot::NodeMetadata{node.op_type(), node.label()});
276 }
277
278 return ActivationSnapshot(std::move(activations), std::move(node_ids), std::move(metadata));
279 }

320 [[nodiscard]] Tensor logit_lens(NodeId node_id) const {
321 auto it = activations_.find(node_id);
322 if (it == activations_.end()) {
323 throw std::invalid_argument(
324 "ExplainerContext::logit_lens: node_id has no cached activation from the most recent "
325 "forward pass");
326 }
327
328 const NodeId head_input_node = static_cast<NodeId>(modules_.size() - 1);
329 auto head_input = activations_.find(head_input_node);
330 if (head_input == activations_.end()) {
331 throw std::invalid_argument(
332 "ExplainerContext::logit_lens: no forward pass has run, so the final module's input "
333 "shape is unknown");
334 }
335 if (!(it->second.shape() == head_input->second.shape())) {
336 throw std::invalid_argument(
337 "ExplainerContext::logit_lens: the node's cached activation shape does not match the "
338 "final module's input shape from the most recent forward pass");
339 }
340
341 Module* head = modules_.back();
342 return head->forward(it->second);
343 }

376 [[nodiscard]] Tensor attention_weights(NodeId node_id) const {
377 if (node_id >= graph_.node_count() || node_id > modules_.size()) {
378 throw std::invalid_argument(
379 "ExplainerContext::attention_weights: node_id is outside the range of nodes the most "
380 "recent forward pass produced (or no forward pass has run yet)");
381 }
382 if (graph_.node(node_id).op_type() != OpType::Attention) {
383 throw std::invalid_argument(
384 "ExplainerContext::attention_weights: node_id does not name an attention layer");
385 }
386
387 auto* mha = dynamic_cast<MultiHeadAttentionModule*>(modules_[node_id - 1]);
388 PULSATRIX_ASSERT(mha != nullptr);
389 return Tensor(mha->last_attention_weights());
390 }

426 [[nodiscard]] CircuitGraph build_circuit_graph(const Tensor& input) {
427 const Tensor baseline = forward_pass(input);
428 const NodeId output_node = output_node_;
429 // Captured before any patching: every patched pass replaces graph_/activations_
430 // wholesale, so the baseline's shapes and per-node metadata are read from this
431 // self-contained copy rather than from state the loop itself overwrites.
432 const ActivationSnapshot clean = activation_snapshot();
433
434 std::vector<CircuitNode> nodes;
435 nodes.reserve(clean.node_ids().size());
436 for (NodeId id : clean.node_ids()) {
437 float ablation_effect = 0.0f;
438 if (id != output_node) {
439 Tensor zero_patch(clean.activation(id));
440 zero_patch.fill(0.0f);
441 const Tensor patched = forward_pass_with_patch(input, id, zero_patch);
442 ablation_effect = l2_distance(baseline, patched);
443 }
444 nodes.push_back(CircuitNode{id, clean.op_type(id), clean.label(id), ablation_effect});
445 }
446
447 std::vector<CircuitEdge> edges;
448 if (!nodes.empty()) {
449 edges.reserve(nodes.size() - 1);
450 for (size_t i = 0; i + 1 < nodes.size(); ++i) {
451 edges.push_back(CircuitEdge{nodes[i].id, nodes[i + 1].id, nodes[i].ablation_effect});
452 }
453 }
454
455 (void)forward_pass(input);
456 return CircuitGraph(std::move(nodes), std::move(edges));
457 }

491 [[nodiscard]] static float l2_distance(const Tensor& a, const Tensor& b) {
492 PULSATRIX_ASSERT(a.numel() == b.numel());
493 float sum_of_squares = 0.0f;
494 for (int64_t i = 0; i < a.numel(); ++i) {
495 const float diff = a.data()[i] - b.data()[i];
496 sum_of_squares += diff * diff;
497 }
498 return std::sqrt(sum_of_squares);
499 }
