72 Tensor forward_pass(const Tensor& input) {
73 last_forward_was_patched_ = false;
74 graph_ = ComputationGraph{};
75 autograd_ = Autograd{};
76 activations_.clear();
77
78 input_node_ = graph_.add_node(OpType::Elementwise, input.shape(), "input");
79 activations_.emplace(input_node_, Tensor(input));
80
81 Tensor current = input;
82 NodeId current_node = input_node_;
83 for (Module* module : modules_) {
84 auto [output, node_id] = module->forward_traced(current, current_node, graph_, autograd_);
85 activations_.emplace(node_id, Tensor(output));
86 current = std::move(output);
87 current_node = node_id;
88 }
89 output_node_ = current_node;
90 return current;
91 }

130 Tensor forward_pass_with_patch(const Tensor& input, NodeId patch_node_id, const Tensor& patch_value) {
131 last_forward_was_patched_ = true;
132 if (patch_node_id > modules_.size()) {
133 throw std::invalid_argument("ExplainerContext::forward_pass_with_patch: patch_node_id out of range");
134 }
135
136 ComputationGraph graph;
137 Autograd autograd;
138 std::unordered_map<NodeId, Tensor> activations;
139
140 NodeId input_node = graph.add_node(OpType::Elementwise, input.shape(), "input");
141
142 Tensor current = input;
143 NodeId current_node = input_node;
144 if (patch_node_id == input_node) {
145 current = check_patch_shape(current, patch_value);
146 }
147 activations.emplace(input_node, Tensor(current));
148
149 for (Module* module : modules_) {
150 auto [output, node_id] = module->forward_traced(current, current_node, graph, autograd);
151 if (node_id == patch_node_id) {
152 output = check_patch_shape(output, patch_value);
153 }
154 activations.emplace(node_id, Tensor(output));
155 current = std::move(output);
156 current_node = node_id;
157 }
158
159 graph_ = std::move(graph);
160 autograd_ = std::move(autograd);
161 activations_ = std::move(activations);
162 input_node_ = input_node;
163 output_node_ = current_node;
164 return current;
165 }

184 Tensor backward_pass(const Tensor& output_grad) {
185 if (last_forward_was_patched_) {
186 throw std::logic_error(
187 "ExplainerContext::backward_pass: the most recent forward pass was "
188 "forward_pass_with_patch(); a patched activation is a constant substitution, not a "
189 "differentiable function of the input, so gradients through it would be silently wrong. "
190 "Run an unpatched forward_pass() before backward_pass().");
191 }
192 autograd_.backward(graph_, output_node_, output_grad);
193 return Tensor(autograd_.gradient(input_node_));
194 }

238 [[nodiscard]] ActivationSnapshot activation_snapshot() const {
239 std::vector<NodeId> node_ids = graph_.topological_order();
240
241 std::unordered_map<NodeId, Tensor> activations;
242 std::unordered_map<NodeId, ActivationSnapshot::NodeMetadata> metadata;
243 for (NodeId id : node_ids) {
244 auto it = activations_.find(id);
245 PULSATRIX_ASSERT(it != activations_.end());
246 activations.emplace(id, Tensor(it->second));
247 const Node& node = graph_.node(id);
248 metadata.emplace(id, ActivationSnapshot::NodeMetadata{node.op_type(), node.label()});
249 }
250
251 return ActivationSnapshot(std::move(activations), std::move(node_ids), std::move(metadata));
252 }

293 [[nodiscard]] Tensor logit_lens(NodeId node_id) const {
294 auto it = activations_.find(node_id);
295 if (it == activations_.end()) {
296 throw std::invalid_argument(
297 "ExplainerContext::logit_lens: node_id has no cached activation from the most recent "
298 "forward pass");
299 }
300
301 const NodeId head_input_node = static_cast<NodeId>(modules_.size() - 1);
302 auto head_input = activations_.find(head_input_node);
303 if (head_input == activations_.end()) {
304 throw std::invalid_argument(
305 "ExplainerContext::logit_lens: no forward pass has run, so the final module's input "
306 "shape is unknown");
307 }
308 if (!(it->second.shape() == head_input->second.shape())) {
309 throw std::invalid_argument(
310 "ExplainerContext::logit_lens: the node's cached activation shape does not match the "
311 "final module's input shape from the most recent forward pass");
312 }
313
314 Module* head = modules_.back();
315 return head->forward(it->second);
316 }

349 [[nodiscard]] Tensor attention_weights(NodeId node_id) const {
350 if (node_id >= graph_.node_count() || node_id > modules_.size()) {
351 throw std::invalid_argument(
352 "ExplainerContext::attention_weights: node_id is outside the range of nodes the most "
353 "recent forward pass produced (or no forward pass has run yet)");
354 }
355 if (graph_.node(node_id).op_type() != OpType::Attention) {
356 throw std::invalid_argument(
357 "ExplainerContext::attention_weights: node_id does not name an attention layer");
358 }
359
360 auto* mha = dynamic_cast<MultiHeadAttentionModule*>(modules_[node_id - 1]);
361 PULSATRIX_ASSERT(mha != nullptr);
362 return Tensor(mha->last_attention_weights());
363 }

399 [[nodiscard]] CircuitGraph build_circuit_graph(const Tensor& input) {
400 const Tensor baseline = forward_pass(input);
401 const NodeId output_node = output_node_;
402 // Captured before any patching: every patched pass replaces graph_/activations_
403 // wholesale, so the baseline's shapes and per-node metadata are read from this
404 // self-contained copy rather than from state the loop itself overwrites.
405 const ActivationSnapshot clean = activation_snapshot();
406
407 std::vector<CircuitNode> nodes;
408 nodes.reserve(clean.node_ids().size());
409 for (NodeId id : clean.node_ids()) {
410 float ablation_effect = 0.0f;
411 if (id != output_node) {
412 Tensor zero_patch(clean.activation(id));
413 zero_patch.fill(0.0f);
414 const Tensor patched = forward_pass_with_patch(input, id, zero_patch);
415 ablation_effect = l2_distance(baseline, patched);
416 }
417 nodes.push_back(CircuitNode{id, clean.op_type(id), clean.label(id), ablation_effect});
418 }
419
420 std::vector<CircuitEdge> edges;
421 if (!nodes.empty()) {
422 edges.reserve(nodes.size() - 1);
423 for (size_t i = 0; i + 1 < nodes.size(); ++i) {
424 edges.push_back(CircuitEdge{nodes[i].id, nodes[i + 1].id, nodes[i].ablation_effect});
425 }
426 }
427
428 (void)forward_pass(input);
429 return CircuitGraph(std::move(nodes), std::move(edges));
430 }

464 [[nodiscard]] static float l2_distance(const Tensor& a, const Tensor& b) {
465 PULSATRIX_ASSERT(a.numel() == b.numel());
466 float sum_of_squares = 0.0f;
467 for (int64_t i = 0; i < a.numel(); ++i) {
468 const float diff = a.data()[i] - b.data()[i];
469 sum_of_squares += diff * diff;
470 }
471 return std::sqrt(sum_of_squares);
472 }
