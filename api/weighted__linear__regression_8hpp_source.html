27inline std::vector<float> SolveLinearSystem(std::vector<std::vector<float>> a, std::vector<float> b) {
28 const auto n = static_cast<int64_t>(b.size());
29 for (int64_t col = 0; col < n; ++col) {
30 int64_t pivot_row = col;
31 float pivot_magnitude = std::fabs(a[static_cast<size_t>(col)][static_cast<size_t>(col)]);
32 for (int64_t row = col + 1; row < n; ++row) {
33 float magnitude = std::fabs(a[static_cast<size_t>(row)][static_cast<size_t>(col)]);
34 if (magnitude > pivot_magnitude) {
35 pivot_magnitude = magnitude;
36 pivot_row = row;
37 }
38 }
39 if (pivot_magnitude < 1e-8f) {
40 throw std::runtime_error("WeightedLinearRegression: near-singular system (degenerate samples?)");
41 }
42 std::swap(a[static_cast<size_t>(col)], a[static_cast<size_t>(pivot_row)]);
43 std::swap(b[static_cast<size_t>(col)], b[static_cast<size_t>(pivot_row)]);
44
45 for (int64_t row = col + 1; row < n; ++row) {
46 float factor = a[static_cast<size_t>(row)][static_cast<size_t>(col)] /
47 a[static_cast<size_t>(col)][static_cast<size_t>(col)];
48 for (int64_t c = col; c < n; ++c) {
49 a[static_cast<size_t>(row)][static_cast<size_t>(c)] -=
50 factor * a[static_cast<size_t>(col)][static_cast<size_t>(c)];
51 }
52 b[static_cast<size_t>(row)] -= factor * b[static_cast<size_t>(col)];
53 }
54 }

57 for (int64_t row = n - 1; row >= 0; --row) {

59 for (int64_t c = row + 1; c < n; ++c) {
60 sum -= a[static_cast<size_t>(row)][static_cast<size_t>(c)] * x[static_cast<size_t>(c)];
61 }
62 x[static_cast<size_t>(row)] = sum / a[static_cast<size_t>(row)][static_cast<size_t>(row)];
63 }
