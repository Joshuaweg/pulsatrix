39inline Configuration DecodeGenotype(const SearchSpace& space, const std::vector<double>& genotype) {
40 if (genotype.size() != space.size()) {
41 throw std::invalid_argument("DecodeGenotype: genotype.size() must equal space.size()");
42 }
43 Configuration config;
44 for (size_t i = 0; i < space.parameters().size(); ++i) {
45 const auto& spec = space.parameters()[i];
46 double gene = genotype[i];
47 if (gene < 0.0 || gene > 1.0) {
48 throw std::invalid_argument("DecodeGenotype: every gene must be in [0, 1]");
49 }
50 switch (spec.kind) {
51 case ParameterKind::Continuous:
52 config[spec.name] = spec.lower + gene * (spec.upper - spec.lower);
53 break;
54 case ParameterKind::LogUniform: {
55 double log_lower = std::log(spec.lower);
56 double log_upper = std::log(spec.upper);
57 config[spec.name] = std::exp(log_lower + gene * (log_upper - log_lower));
58 break;
59 }
60 case ParameterKind::Integer: {
61 double interpolated = spec.lower + gene * (spec.upper - spec.lower);
62 int64_t rounded = static_cast<int64_t>(std::round(interpolated));
63 rounded = std::min(rounded, static_cast<int64_t>(spec.upper));
64 rounded = std::max(rounded, static_cast<int64_t>(spec.lower));
65 config[spec.name] = rounded;
66 break;
67 }
68 case ParameterKind::Categorical: {
69 size_t num_categories = spec.categories.size();
70 size_t index = static_cast<size_t>(gene * static_cast<double>(num_categories));
71 index = std::min(index, num_categories - 1);
72 config[spec.name] = spec.categories[index];
73 break;
74 }
