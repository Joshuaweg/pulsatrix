26inline double BinomialCoefficient(int64_t n, int64_t k) {
27 double result = 1.0;
28 for (int64_t i = 1; i <= k; ++i) {
29 result = result * static_cast<double>(n - k + i) / static_cast<double>(i);
30 }
31 return result;
32}

39inline float ShapKernelWeight(int64_t n, int64_t coalition_size) {
40 double denominator =
41 BinomialCoefficient(n, coalition_size) * static_cast<double>(coalition_size) * static_cast<double>(n - coalition_size);
42 return static_cast<float>(static_cast<double>(n - 1) / denominator);
43}
