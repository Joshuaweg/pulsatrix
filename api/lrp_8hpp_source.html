76 [[nodiscard]] Attribution explain(ExplainerContext& ctx, const Tensor& input, const LRPTarget& target,
77 DeviceBackend* backend) const {
78 Tensor output = ctx.forward_pass(input);
79 if (output.rank() != 2) {
80 throw std::invalid_argument("LRP::explain: network output must be rank-2 (N, num_classes)");
81 }
82 const int64_t N = output.shape().dim(0);
83 const int64_t C = output.shape().dim(1);
84 const std::vector<int64_t> targets = per_row(target.targets, N, C, "targets");
85 const std::vector<int64_t> contrasts =
86 target.contrasts.empty() ? std::vector<int64_t>{} : per_row(target.contrasts, N, C, "contrasts");
87 for (size_t n = 0; n < contrasts.size(); ++n) {
88 if (contrasts[n] == targets[n]) {
89 throw std::invalid_argument(
90 "LRP::explain: a row's contrast equals its target (the seed would be zero)");
91 }
92 }
93
94 // Seed on the host, then upload: valid whatever device the output lives on.
95 std::vector<float> y;
96 if (target.seed == LRPSeed::OutputValue) {
97 y.resize(static_cast<size_t>(N * C));
98 output.backend()->copy(y.data(), output.data(), y.size() * sizeof(float),
99 output.device() == DeviceType::Cpu ? CopyDirection::HostToHost
100 : CopyDirection::DeviceToHost);
101 }
102 auto seed_value = [&](int64_t n, int64_t c) {
103 return target.seed == LRPSeed::OutputValue ? y[static_cast<size_t>(n * C + c)] : 1.0f;
104 };
105 std::vector<float> seed(static_cast<size_t>(N * C), 0.0f);
106 float relevance_out_sum = 0.0f;
107 for (int64_t n = 0; n < N; ++n) {
108 const int64_t t = targets[static_cast<size_t>(n)];
109 seed[static_cast<size_t>(n * C + t)] = seed_value(n, t);
110 relevance_out_sum += seed_value(n, t);
111 if (!contrasts.empty()) {
112 const int64_t c = contrasts[static_cast<size_t>(n)];
113 seed[static_cast<size_t>(n * C + c)] = -seed_value(n, c);
114 relevance_out_sum -= seed_value(n, c);
115 }
116 }
117 Tensor seed_tensor(output.shape(), backend, seed, output.device());
118
119 Tensor relevance = ctx.relevance_pass(seed_tensor, config_);
120 const float relevance_in_sum =
121 relevance.backend()->sum(relevance.data(), static_cast<size_t>(relevance.numel()));
122
123 return Attribution{"lrp",
124 std::move(relevance),
125 {{"rule", "epsilon"},
126 {"epsilon", std::to_string(config_.epsilon)},
127 {"seed", target.seed == LRPSeed::OutputValue ? "output_value" : "one_hot"},
128 {"targets", join(target.targets)},
129 {"contrasts", join(target.contrasts)},
130 {"relevance_out_sum", std::to_string(relevance_out_sum)},
131 {"relevance_in_sum", std::to_string(relevance_in_sum)}}};
132 }

137 // Expands a 1-entry list to N rows and validates every index against C classes.
138 static std::vector<int64_t> per_row(const std::vector<int64_t>& indices, int64_t N, int64_t C,
139 const char* what) {
140 if (indices.size() != 1 && static_cast<int64_t>(indices.size()) != N) {
141 throw std::invalid_argument(std::string("LRP::explain: ") + what +
142 " must have 1 entry or one per row of the output");
143 }
144 std::vector<int64_t> rows(static_cast<size_t>(N));
145 for (int64_t n = 0; n < N; ++n) {
146 const int64_t index = indices.size() == 1 ? indices[0] : indices[static_cast<size_t>(n)];
147 if (index < 0 || index >= C) {
148 throw std::invalid_argument(std::string("LRP::explain: ") + what + " index out of range");
149 }
150 rows[static_cast<size_t>(n)] = index;
151 }
152 return rows;
153 }

155 static std::string join(const std::vector<int64_t>& v) {
156 std::string out;
157 for (size_t i = 0; i < v.size(); ++i) {
158 out += (i == 0 ? "" : ",") + std::to_string(v[i]);
159 }
160 return out;
161 }
