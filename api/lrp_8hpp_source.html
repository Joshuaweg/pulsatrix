214 [[nodiscard]] Attribution explain(ExplainerContext& ctx, const Tensor& input, const LRPTarget& target,
215 DeviceBackend* backend) const {
216 Tensor output = ctx.forward_pass(input);
217 if (output.rank() != 2) {
218 throw std::invalid_argument("LRP::explain: network output must be rank-2 (N, num_classes)");
219 }
220 const int64_t N = output.shape().dim(0);
221 const int64_t C = output.shape().dim(1);
222 const std::vector<int64_t> targets = per_row(target.targets, N, C, "targets");
223 const std::vector<int64_t> contrasts =
224 target.contrasts.empty() ? std::vector<int64_t>{} : per_row(target.contrasts, N, C, "contrasts");
225 for (size_t n = 0; n < contrasts.size(); ++n) {
226 if (contrasts[n] == targets[n]) {
227 throw std::invalid_argument(
228 "LRP::explain: a row's contrast equals its target (the seed would be zero)");
229 }
230 }
231
232 // Seed on the host, then upload: valid whatever device the output lives on.
233 std::vector<float> y;
234 if (target.seed == LRPSeed::OutputValue) {
235 y.resize(static_cast<size_t>(N * C));
236 output.backend()->copy(y.data(), output.data(), y.size() * sizeof(float),
237 output.device() == DeviceType::Cpu ? CopyDirection::HostToHost
238 : CopyDirection::DeviceToHost);
239 }
240 auto seed_value = [&](int64_t n, int64_t c) {
241 return target.seed == LRPSeed::OutputValue ? y[static_cast<size_t>(n * C + c)] : 1.0f;
242 };
243 std::vector<float> seed(static_cast<size_t>(N * C), 0.0f);
244 float relevance_out_sum = 0.0f;
245 for (int64_t n = 0; n < N; ++n) {
246 const int64_t t = targets[static_cast<size_t>(n)];
247 seed[static_cast<size_t>(n * C + t)] = seed_value(n, t);
248 relevance_out_sum += seed_value(n, t);
249 if (!contrasts.empty()) {
250 const int64_t c = contrasts[static_cast<size_t>(n)];
251 seed[static_cast<size_t>(n * C + c)] = -seed_value(n, c);
252 relevance_out_sum -= seed_value(n, c);
253 }
254 }
255 Tensor seed_tensor(output.shape(), backend, seed, output.device());
256
257 std::vector<LRPRuleConfig> configs;
258 configs.reserve(ctx.modules().size());
259 for (size_t i = 0; i < ctx.modules().size(); ++i) {
260 configs.push_back(composite_ ? composite_(i, *ctx.modules()[i]) : config_);
261 }
262 std::string rules;
263 for (size_t i = 0; i < configs.size(); ++i) {
264 rules += (i == 0 ? "" : ",") + lrp_rule_name(configs[i].rule);
265 }
266 Tensor relevance = ctx.relevance_pass(seed_tensor, configs);
267 const float relevance_in_sum =
268 relevance.backend()->sum(relevance.data(), static_cast<size_t>(relevance.numel()));
269
270 return Attribution{"lrp",
271 std::move(relevance),
272 {{"rule", composite_ ? "composite:" + composite_name_ : lrp_rule_name(config_.rule)},
273 {"rules", rules},
274 {"epsilon", std::to_string(config_.epsilon)},
275 {"seed", target.seed == LRPSeed::OutputValue ? "output_value" : "one_hot"},
276 {"targets", join(target.targets)},
277 {"contrasts", join(target.contrasts)},
278 {"relevance_out_sum", std::to_string(relevance_out_sum)},
279 {"relevance_in_sum", std::to_string(relevance_in_sum)}}};
280 }

289 // Expands a 1-entry list to N rows and validates every index against C classes.
290 static std::vector<int64_t> per_row(const std::vector<int64_t>& indices, int64_t N, int64_t C,
291 const char* what) {
292 if (indices.size() != 1 && static_cast<int64_t>(indices.size()) != N) {
293 throw std::invalid_argument(std::string("LRP::explain: ") + what +
294 " must have 1 entry or one per row of the output");
295 }
296 std::vector<int64_t> rows(static_cast<size_t>(N));
297 for (int64_t n = 0; n < N; ++n) {
298 const int64_t index = indices.size() == 1 ? indices[0] : indices[static_cast<size_t>(n)];
299 if (index < 0 || index >= C) {
300 throw std::invalid_argument(std::string("LRP::explain: ") + what + " index out of range");
301 }
302 rows[static_cast<size_t>(n)] = index;
303 }
304 return rows;
305 }

307 static std::string join(const std::vector<int64_t>& v) {
308 std::string out;
309 for (size_t i = 0; i < v.size(); ++i) {
310 out += (i == 0 ? "" : ",") + std::to_string(v[i]);
311 }
312 return out;
313 }
